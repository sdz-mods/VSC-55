cmake_minimum_required(VERSION 3.23)
project(vsc55 LANGUAGES C CXX RC)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build configuration" FORCE)
endif()
option(CMAKE_INTERPROCEDURAL_OPTIMIZATION "Enable link-time optimization" ON)
set(VSC55_CPU_PROFILE "base" CACHE STRING "Win98 CPU profile: base, sse, sse2")
set_property(CACHE VSC55_CPU_PROFILE PROPERTY STRINGS base sse sse2)
set(default_build_profile "${VSC55_CPU_PROFILE}")
if(CMAKE_INTERPROCEDURAL_OPTIMIZATION)
  string(APPEND default_build_profile "-lto")
endif()
set(VSC55_BUILD_PROFILE "${default_build_profile}" CACHE STRING "Build label recorded in diagnostics")
if(CMAKE_INTERPROCEDURAL_OPTIMIZATION)
  include(CheckIPOSupported)
  check_ipo_supported(RESULT ipo_ok OUTPUT ipo_error LANGUAGES C CXX)
  if(NOT ipo_ok)
    message(FATAL_ERROR "LTO unavailable: ${ipo_error}")
  endif()
endif()
set(VSC55_CPU_OPTIONS -march=i686 -mtune=generic)
if(VSC55_CPU_PROFILE STREQUAL "sse")
  list(APPEND VSC55_CPU_OPTIONS -msse -mno-sse2 -mfpmath=sse)
elseif(VSC55_CPU_PROFILE STREQUAL "sse2")
  list(APPEND VSC55_CPU_OPTIONS -msse2 -mfpmath=sse)
elseif(NOT VSC55_CPU_PROFILE STREQUAL "base")
  message(FATAL_ERROR "Unknown CPU profile: ${VSC55_CPU_PROFILE}")
endif()
set(UPSTREAM "${CMAKE_CURRENT_SOURCE_DIR}/upstream/Nuked-SC55")
set(NUKED_SOURCE "VSC-55 / Nuked-SC55 0.7.0 / upstream 02f6e3d7bad89af33514bd48211bb950f8ad0e6b")
set(NUKED_ENABLE_ASIO OFF)
# Keep the original decoder: cached decoder must pass exact-output checks first.
set(NUKED_ENABLE_DECODER2 OFF)
configure_file("${UPSTREAM}/src/backend/config.h.in" generated/config.h)
file(GLOB BACKEND CONFIGURE_DEPENDS "${UPSTREAM}/src/backend/*.cpp")
add_library(sc55backend STATIC ${BACKEND} "${UPSTREAM}/src/backend/sha/sha224-256.c")
target_include_directories(sc55backend PUBLIC "${UPSTREAM}/src/backend" "${CMAKE_CURRENT_BINARY_DIR}/generated")
add_executable(coreprobe src/probe/coreprobe.cpp)
target_link_libraries(coreprobe PRIVATE sc55backend)
add_executable(stress src/probe/stress.cpp)
target_link_libraries(stress PRIVATE sc55backend)
add_executable(samplefifo_test tests/samplefifo.cpp)
target_link_libraries(samplefifo_test PRIVATE sc55backend)
add_executable(dcoutput_test tests/dcoutput.cpp)
target_link_libraries(dcoutput_test PRIVATE sc55backend)
add_executable(mute_test tests/mute.cpp)
target_link_libraries(mute_test PRIVATE sc55backend)
add_executable(wavplay src/probe/wavplay.c)
target_link_libraries(wavplay PRIVATE winmm)
option(VSC55_TIMING_DIAGNOSTICS "Enable detailed worker/receiver timing instrumentation" OFF)
set_source_files_properties(src/gui/resource.rc PROPERTIES OBJECT_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/src/gui/vsc55.ico")
add_executable(vsc55 src/host/vsc55.cpp src/gui/resource.rc)
if(VSC55_TIMING_DIAGNOSTICS)
  target_compile_definitions(vsc55 PRIVATE VSC55_TIMING_DIAGNOSTICS=1)
endif()
target_link_libraries(vsc55 PRIVATE sc55backend winmm)
foreach(t vsc55 stress coreprobe)
  target_compile_definitions(${t} PRIVATE VSC55_BUILD_PROFILE="${VSC55_BUILD_PROFILE}")
endforeach()
add_executable(miditone tests/miditone.c)
target_link_libraries(miditone PRIVATE winmm)
add_executable(midirecv src/host/midirecv.c)
add_executable(miditest tests/miditest.c)
target_link_libraries(miditest PRIVATE winmm)
add_executable(midireg installer/midisetup.c)
target_link_libraries(midireg PRIVATE user32 advapi32)
add_executable(vsccfg WIN32 src/gui/vsccfg.c src/gui/frontpanel.c src/gui/lcdview.cpp src/gui/resource.rc)
target_link_libraries(vsccfg PRIVATE sc55backend winmm comctl32 shell32 ole32 gdi32)
if(MINGW)
  add_link_options(-static)
  target_link_options(vsc55 PRIVATE -static)
  target_link_options(vsccfg PRIVATE -static)
  # Win98 USER/SHELL thunks need committed stack headroom (64 KiB).
  target_link_options(vsccfg PRIVATE "SHELL:-Xlinker --stack -Xlinker 1048576,65536")
  target_link_options(miditone PRIVATE -static)
  target_link_options(coreprobe PRIVATE -static)
  target_link_options(stress PRIVATE -static)
  target_link_options(samplefifo_test PRIVATE -static)
  target_link_options(dcoutput_test PRIVATE -static)
  target_link_options(wavplay PRIVATE -static)
  target_link_options(midirecv PRIVATE -static)
  target_link_options(miditest PRIVATE -static)
  target_link_options(midireg PRIVATE -static)
endif()
if(VSC55_WIN98)
  add_compile_definitions(WINVER=0x0410 _WIN32_WINNT=0x0410)
  foreach(t sc55backend coreprobe stress wavplay midirecv miditest midireg vsc55 miditone vsccfg)
    target_compile_definitions(${t} PRIVATE WINVER=0x0410 _WIN32_WINNT=0x0410)
    target_compile_options(${t} PRIVATE ${VSC55_CPU_OPTIONS})
  endforeach()
  foreach(t coreprobe stress wavplay midirecv miditest midireg vsc55 miditone vsccfg)
    target_link_options(${t} PRIVATE -Wl,--major-os-version,4,--minor-os-version,0,--major-subsystem-version,4,--minor-subsystem-version,0)
  endforeach()
endif()
